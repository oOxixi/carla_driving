cmake_minimum_required(VERSION 3.18)
project(student_nashp_runtime LANGUAGES CXX)
set(UCP_ROOT "" CACHE PATH "OE samples/ucp_tutorial/deps_aarch64/ucp")
find_path(UCP_INCLUDE hobot/dnn/hb_dnn.h PATHS "${UCP_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
find_library(DNN_LIBRARY dnn PATHS "${UCP_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
find_library(UCP_LIBRARY hbucp PATHS "${UCP_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
add_executable(student_hbdnn src/student_hbdnn.cpp)
target_compile_features(student_hbdnn PRIVATE cxx_std_17)
target_compile_options(student_hbdnn PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(student_hbdnn PRIVATE "${UCP_INCLUDE}")
target_link_libraries(student_hbdnn PRIVATE "${DNN_LIBRARY}" "${UCP_LIBRARY}")
set_target_properties(student_hbdnn PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
install(TARGETS student_hbdnn RUNTIME DESTINATION .)
